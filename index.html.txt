    // ТВОЙ РАБОЧИЙ GOOGLE ШЛЮЗ
    const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxVsFy3G9JPkFhLRfIqX-cEEPvJeuwMd3oqiYwHjaFQ7qdFNbnRKiAOTivgWc33kvxdeA/exec";

    let bookingData = { service: '', price: '', date: '', time: '', name: '', phone: '' };
    let busySlotsGlobal = [];

    // Загрузка занятых слотов
    async function loadBusySlots() {
      try {
        let response = await fetch(GOOGLE_SCRIPT_URL);
        let resData = await response.json();
        busySlotsGlobal = resData.busy || [];
      } catch (e) { console.error("Ошибка загрузки:", e); }
    }
    loadBusySlots();

    function selectService(element, name, price) {
      document.querySelectorAll('.service-card').forEach(el => el.classList.remove('active'));
      element.classList.add('active');
      bookingData.service = name;
      bookingData.price = price;
      
      document.getElementById('step-datetime').classList.remove('hidden');
      generateCalendar();
      document.getElementById('step-datetime').scrollIntoView({ behavior: 'smooth' });
    }

    function generateCalendar() {
      const container = document.getElementById('dates-container');
      container.innerHTML = '';
      document.getElementById('slots-container').innerHTML = '';
      
      let current = new Date();
      let addedDays = 0;

      while (addedDays < 7) {
        current.setDate(current.getDate() + 1);
        let dayOfWeek = current.getDay();
        
        // 1 - Понедельник, 2 - Вторник (Твои законные выходные)
        if (dayOfWeek === 1 || dayOfWeek === 2) continue;

        let year = current.getFullYear();
        let month = String(current.getMonth() + 1).padStart(2, '0');
        let day = String(current.getDate()).padStart(2, '0');
        let fullDate = year + "-" + month + "-" + day;
        
        let daysList = ['вс', 'пн', 'вт', 'ср', 'чт', 'пт', 'сб'];
        let visualDate = day + "." + month + " (" + daysList[dayOfWeek] + ")";

        let btn = document.createElement('div');
        btn.className = 'btn-choice';
        btn.innerText = visualDate;
        btn.onclick = function() {
          document.querySelectorAll('#dates-container .btn-choice').forEach(el => el.classList.remove('active'));
          btn.classList.add('active');
          selectDate(fullDate);
        };
        container.appendChild(btn);
        addedDays++;
      }
    }

    function selectDate(dateStr) {
      bookingData.date = dateStr;
      const container = document.getElementById('slots-container');
      container.innerHTML = '';
      
      // Рабочие часы по твоему бизнес-плану (с 10:00 до 20:00)
      const allSlots = ["10:00", "11:00", "12:00", "13:00", "14:00", "15:00", "16:00", "17:00", "18:00", "19:00"];
      
      allSlots.forEach(slot => {
        let btn = document.createElement('div');
        btn.className = 'btn-choice';
        btn.innerText = slot;
        
        // Проверяем занятость по базе данных
